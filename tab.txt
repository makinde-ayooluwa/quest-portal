fetch("https://docs.google.com/spreadsheets/d/SHEET_ID/export?format=csv")
  .then(res => res.text())
  .then(csv => {
    const rows = parseCSV(csv);

    const headerRow1 = rows[0]; // NAME | MATH | ""
    const headerRow2 = rows[1]; // ""   | 1st  | 2nd

    // Fix merged subject headers
    let lastSubject = "";
    const subjectsByColumn = headerRow1.map((h, i) => {
      if (h) lastSubject = h;
      return {
        subject: lastSubject,
        term: headerRow2[i]
      };
    });

    // Process student rows
    const students = rows.slice(2).map(row => {
      const student = {
        Fullname: row[0],
        subjects: []
      };

      subjectsByColumn.forEach((col, i) => {
        if (i === 0 || !col.subject || !col.term) return;

        let subject = student.subjects.find(s => s.name === col.subject);
        if (!subject) {
          subject = { name: col.subject };
          student.subjects.push(subject);
        }

        subject[`${col.term} Term`] = Number(row[i]);
      });

      return student;
    });

    console.log(students);
  });

function parseCSV(text) {
  return text
    .trim()
    .split("\n")
    .map(row => row.split(",").map(c => c.trim()));
}
